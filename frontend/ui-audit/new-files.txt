===== frontend/src/core/theme/themeStore.js =====
const STORAGE_KEY = 'um-theme';
const THEMES = Object.freeze({ LIGHT: 'light', DARK: 'dark' });
const listeners = new Set();
let releaseFrame;
let paintFrame;
let explicitPreference = false;

const validTheme = (value) => value === 'light' || value === 'dark';
const systemTheme = () => window.matchMedia('(prefers-color-scheme: dark)').matches ? 'dark' : 'light';

export const getTheme = () => typeof document === 'undefined' ? 'light'
    : document.documentElement.classList.contains('dark') ? 'dark' : 'light';

export function subscribeTheme(listener) {
    listeners.add(listener);
    return () => listeners.delete(listener);
}

function applyTheme(theme) {
    const root = document.documentElement;
    cancelAnimationFrame(paintFrame);
    cancelAnimationFrame(releaseFrame);
    root.classList.add('no-transitions');
    root.classList.toggle('dark', theme === 'dark');
    root.classList.toggle('light', theme === 'light');
    root.dataset.theme = theme;
    root.style.colorScheme = theme;
    listeners.forEach(listener => listener());
    // Suppress through the first painted frame without a synchronous layout read.
    paintFrame = requestAnimationFrame(() => {
        releaseFrame = requestAnimationFrame(() => root.classList.remove('no-transitions'));
    });
}

function setTheme(theme) {
    if (!validTheme(theme)) return;
    explicitPreference = true;
    if (getTheme() !== theme) applyTheme(theme);
    try { localStorage.setItem(STORAGE_KEY, theme); } catch { /* Storage may be blocked. */ }
}

export const themeActions = Object.freeze({
    THEMES,
    setTheme,
    toggleTheme: () => setTheme(getTheme() === 'dark' ? 'light' : 'dark'),
    resetTheme: () => {
        explicitPreference = false;
        try { localStorage.removeItem(STORAGE_KEY); } catch { /* Storage may be blocked. */ }
        applyTheme(systemTheme());
    },
});

export function startThemeSync() {
    let stored;
    try { stored = localStorage.getItem(STORAGE_KEY); } catch { /* Storage may be blocked. */ }
    explicitPreference = validTheme(stored);
    const initial = explicitPreference ? stored : systemTheme();
    if (getTheme() !== initial) applyTheme(initial);
    const media = window.matchMedia('(prefers-color-scheme: dark)');
    const onSystem = () => { if (!explicitPreference) applyTheme(systemTheme()); };
    const onStorage = (event) => {
        if (event.key !== STORAGE_KEY && event.key !== null) return;
        explicitPreference = validTheme(event.newValue);
        applyTheme(explicitPreference ? event.newValue : systemTheme());
    };
    media.addEventListener('change', onSystem);
    window.addEventListener('storage', onStorage);
    return () => {
        media.removeEventListener('change', onSystem);
        window.removeEventListener('storage', onStorage);
    };
}

===== frontend/src/shared/components/ui/Form/Form.jsx =====
import { forwardRef, useId, useRef, useEffect, useCallback } from 'react';

const Form = forwardRef(({ children, className, onBlurCapture, onChangeCapture, onInvalidCapture, ...props }, forwardedRef) => {
    const formRef = useRef(null);
    const feedback = useRef(new Map());
    const formId = useId();
    const nextFieldId = useRef(0);
    const nextFeedbackId = useRef(0);
    const setRef = useCallback(node => {
        formRef.current = node;
        if (typeof forwardedRef === 'function') forwardedRef(node);
        else if (forwardedRef) forwardedRef.current = node;
    }, [forwardedRef]);

    const validate = (field) => {
        if (!field.matches?.('input, select, textarea') || field.disabled || field.readOnly || !field.willValidate) return;
        let entry = feedback.current.get(field);
        // Existing domain errors remain owned by the field's component.
        if (!entry && field.getAttribute('aria-invalid') === 'true') return;
        if (!entry) {
            const message = document.createElement('p');
            message.id = `${formId}-feedback-${nextFeedbackId.current++}`;
            message.className = 'form-field-error';
            message.setAttribute('aria-live', 'polite');
            entry = { message, describedBy: field.getAttribute('aria-describedby'), invalid: field.getAttribute('aria-invalid') };
            feedback.current.set(field, entry);
            const anchor = field.closest('label') || (field.parentElement.classList.contains('relative') ? field.parentElement : field);
            anchor.insertAdjacentElement('afterend', message);
        }
        const invalid = !field.validity.valid;
        entry.message.textContent = invalid ? field.validationMessage : '';
        entry.message.hidden = !invalid;
        field.setAttribute('aria-invalid', String(invalid));
        const ids = [entry.describedBy, invalid ? entry.message.id : null].filter(Boolean).join(' ');
        if (ids) field.setAttribute('aria-describedby', ids);
        else field.removeAttribute('aria-describedby');
    };

    useEffect(() => {
        const form = formRef.current;
        const linkLabels = () => {
            feedback.current.forEach((entry, field) => {
                if (!form.contains(field)) { entry.message.remove(); feedback.current.delete(field); }
            });
            form.querySelectorAll('label:not([for])').forEach(label => {
                const selector = 'input:not([type="hidden"]), select, textarea, button[aria-haspopup="listbox"]';
                const sibling = label.nextElementSibling;
                const field = label.querySelector(selector)
                    || (sibling?.matches(selector) ? sibling : sibling?.querySelector(selector))
                    || label.parentElement.querySelector(selector);
                if (!field) return;
                if (!field.id) field.id = `${formId}-field-${nextFieldId.current++}`;
                label.htmlFor = field.id;
            });
        };
        linkLabels();
        const observer = new MutationObserver(linkLabels);
        observer.observe(form, { childList: true, subtree: true });
        const entries = feedback.current;
        return () => {
            observer.disconnect();
            entries.forEach(({ message, describedBy, invalid }, field) => {
                message.remove();
                if (describedBy === null) field.removeAttribute('aria-describedby');
                else field.setAttribute('aria-describedby', describedBy);
                if (invalid === null) field.removeAttribute('aria-invalid');
                else field.setAttribute('aria-invalid', invalid);
            });
            entries.clear();
        };
    }, [formId]);

    return (
        <form
            {...props}
            className={['ui-form', className].filter(Boolean).join(' ')}
            ref={setRef}
            onBlurCapture={event => { onBlurCapture?.(event); if (!event.defaultPrevented) validate(event.target); }}
            onChangeCapture={event => {
                onChangeCapture?.(event);
                if (!event.defaultPrevented && feedback.current.has(event.target)) validate(event.target);
            }}
            onInvalidCapture={event => { onInvalidCapture?.(event); if (!event.defaultPrevented) validate(event.target); }}
        >
            {children}
        </form>
    );
});

Form.displayName = 'Form';
export default Form;

import { Sun, Moon } from 'lucide-react';
import { useTheme } from '@/app/providers/ThemeProvider';
import Button from '../Button/Button';

const ThemeToggle = () => {
    const { isDark, toggleTheme } = useTheme();
    return (
        <Button
            variant="ghost"
            iconOnly
            onClick={toggleTheme}
            aria-label={`Switch to ${isDark ? 'light' : 'dark'} mode`}
            aria-pressed={isDark}
            className="h-11 w-11"
        >
            {isDark ? <Sun className="h-5 w-5" /> : <Moon className="h-5 w-5" />}
        </Button>
    );
};

export default ThemeToggle;

===== frontend/tests/unit/themeControls.test.jsx =====
import { describe, it, expect, vi, afterEach } from 'vitest';
import { render, screen, fireEvent, waitFor } from '@testing-library/react';
import ThemeProvider from '@/app/providers/ThemeProvider';
import ThemeToggle from '@/shared/components/ui/ThemeToggle/ThemeToggle';
import Form from '@/shared/components/ui/Form/Form';
import Input from '@/shared/components/ui/Input/Input';
import Modal from '@/shared/components/ui/Modal/Modal';
import IconSelect from '@/shared/components/ui/IconSelect/IconSelect';
import { themeActions } from '@/core/theme/themeStore';

afterEach(() => {
    localStorage.clear();
    document.documentElement.className = 'light';
    document.documentElement.dataset.theme = 'light';
    vi.restoreAllMocks();
});

describe('theme switching', () => {
    it('updates only subscribers, persists rapid toggles, and releases suppression', async () => {
        let renders = 0;
        function Content() { renders++; return <p>Stable content</p>; }
        render(<ThemeProvider><ThemeToggle /><Content /></ThemeProvider>);
        const before = renders;
        for (let i = 0; i < 10; i++) fireEvent.click(screen.getByRole('button'));
        expect(renders).toBe(before);
        expect(document.documentElement.dataset.theme).toBe('light');
        expect(localStorage.getItem('um-theme')).toBe('light');
        await waitFor(() => expect(document.documentElement).not.toHaveClass('no-transitions'));
    });

    it('syncs valid preferences and storage.clear across tabs', () => {
        render(<ThemeProvider><ThemeToggle /></ThemeProvider>);
        fireEvent(window, new StorageEvent('storage', { key: 'um-theme', newValue: 'dark' }));
        expect(screen.getByRole('button')).toHaveAttribute('aria-pressed', 'true');
        fireEvent(window, new StorageEvent('storage', { key: null }));
        expect(document.documentElement.style.colorScheme).toBe('light');
    });

    it('keeps switching when persistence is unavailable', () => {
        vi.spyOn(Storage.prototype, 'setItem').mockImplementation(() => { throw new Error('blocked'); });
        expect(() => themeActions.setTheme('dark')).not.toThrow();
        expect(document.documentElement).toHaveClass('dark');
    });
});

describe('form accessibility', () => {
    it('shows native constraint errors on blur and clears them on change', () => {
        render(<Form><label>Email<input type="email" required /></label></Form>);
        const field = screen.getByRole('textbox', { name: 'Email' });
        expect(field).not.toHaveAttribute('aria-invalid');
        fireEvent.blur(field);
        expect(field).toHaveAttribute('aria-invalid', 'true');
        const feedbackId = field.getAttribute('aria-describedby');
        expect(document.getElementById(feedbackId)).toBeVisible();
        fireEvent.change(field, { target: { value: 'member@example.com' } });
        expect(field).toHaveAttribute('aria-invalid', 'false');
        expect(document.getElementById(feedbackId)).not.toBeVisible();
    });

    it('keeps generated field IDs and described messages stable on rerender', () => {
        const { rerender } = render(<Input label="Amount" error="Required" aria-describedby="details" />);
        const field = screen.getByRole('textbox');
        const id = field.id;
        expect(field).toHaveAttribute('aria-describedby', `details ${id}-error`);
        rerender(<Input label="Amount" success="Saved" aria-describedby="details" />);
        expect(field.id).toBe(id);
        expect(field).toHaveAttribute('aria-describedby', `details ${id}-success`);
    });

    it('links dynamically added labels with unique field IDs', async () => {
        const { rerender } = render(<Form><div><label>First</label><input /></div></Form>);
        const firstId = screen.getByRole('textbox', { name: 'First' }).id;
        rerender(<Form><div><label>First</label><input /></div><div><label>Second</label><input /></div></Form>);
        await waitFor(() => expect(screen.getByRole('textbox', { name: 'Second' }).id).not.toBe(firstId));
    });
});

describe('dialog focus', () => {
    it('autofocuses the first enabled field and traps Tab around enabled controls', async () => {
        render(<Modal isOpen title="Edit" onClose={vi.fn()} footer={<button>Save</button>}>
            <input disabled aria-label="Disabled" /><input aria-label="Name" />
        </Modal>);
        const name = screen.getByRole('textbox', { name: 'Name' });
        await waitFor(() => expect(name).toHaveFocus());
        screen.getByRole('button', { name: 'Save' }).focus();
        fireEvent.keyDown(document, { key: 'Tab' });
        expect(screen.getByRole('button', { name: 'Close dialog' })).toHaveFocus();
    });

    it('preserves field focus during a parent rerender and restores the opener on close', async () => {
        const opener = document.createElement('button');
        document.body.append(opener);
        opener.focus();
        const { rerender, unmount } = render(<Modal isOpen title="Edit" onClose={() => {}}><input aria-label="Name" /></Modal>);
        await waitFor(() => expect(screen.getByRole('textbox')).toHaveFocus());
        rerender(<Modal isOpen title="Edit" onClose={() => {}}><input aria-label="Name" /></Modal>);
        expect(screen.getByRole('textbox')).toHaveFocus();
        unmount();
        expect(opener).toHaveFocus();
        opener.remove();
    });

    it('uses unique labels when dialogs are nested and only closes the top dialog', () => {
        const outerClose = vi.fn(), innerClose = vi.fn();
        render(<><Modal isOpen title="Outer" onClose={outerClose}><p>Outer body</p></Modal><Modal isOpen title="Inner" onClose={innerClose}><p>Inner body</p></Modal></>);
        const dialogs = screen.getAllByRole('dialog');
        expect(dialogs[0].getAttribute('aria-labelledby')).not.toBe(dialogs[1].getAttribute('aria-labelledby'));
        fireEvent.keyDown(document, { key: 'Escape' });
        expect(innerClose).toHaveBeenCalledOnce();
        expect(outerClose).not.toHaveBeenCalled();
    });

    it('retains the page scroll lock when a select closes inside a dialog', () => {
        const { unmount } = render(<Modal isOpen title="Edit" onClose={vi.fn()}>
            <IconSelect name="month" value="1" options={[{ value: '1', label: 'January' }]} onChange={vi.fn()} />
        </Modal>);
        fireEvent.click(screen.getByRole('button', { name: 'January' }));
        fireEvent.keyDown(document, { key: 'Escape' });
        expect(document.body.style.overflow).toBe('hidden');
        unmount();
        expect(document.body.style.overflow).toBe('');
    });
});
